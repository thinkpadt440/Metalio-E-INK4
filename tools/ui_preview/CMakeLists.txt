cmake_minimum_required(VERSION 3.16)
project(product_ui_preview C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVROOT ${ROOT}/managed_components/lvgl__lvgl)
file(GLOB_RECURSE LVFILES ${LVROOT}/src/*.c)
add_library(preview_lvgl STATIC ${LVFILES})
target_compile_definitions(preview_lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE LV_KCONFIG_IGNORE)
target_include_directories(preview_lvgl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${LVROOT} PRIVATE ${ROOT}/main/display)
add_executable(preview preview.cc ${ROOT}/main/display/screen/dashboard_screen/dashboard_screen.cc ${ROOT}/main/display/screen/harness_screen/harness_screen.cc ${ROOT}/main/display/font/fontpack_lvgl.cc ${ROOT}/main/display/font/font_loader.c ${ROOT}/main/display/font/font_misans_regular_160_2.c)
target_include_directories(preview BEFORE PRIVATE stubs)
target_include_directories(preview PRIVATE ${ROOT}/main ${ROOT}/main/display ${ROOT}/main/display/font ${ROOT}/main/display/screen)
target_compile_definitions(preview PRIVATE PREVIEW_ROOT="${ROOT}")
target_compile_options(preview PRIVATE -UNDEBUG)
target_link_libraries(preview PRIVATE preview_lvgl m)
