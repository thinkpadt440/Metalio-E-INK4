cmake_minimum_required(VERSION 3.16)
project(ws_transport_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
set(CJSON_ROOT "${CJSON_ROOT}" CACHE PATH "ESP-IDF components/json/cJSON directory")
if(NOT EXISTS "${CJSON_ROOT}/cJSON.c")
 set(CJSON_ROOT "$ENV{IDF_PATH}/components/json/cJSON")
endif()
if(NOT EXISTS "${CJSON_ROOT}/cJSON.c")
 message(FATAL_ERROR "Set CJSON_ROOT to the ESP-IDF components/json/cJSON directory")
endif()
add_executable(ws_transport_tests websocket_test.cc ${ROOT}/components/esp-ml307/src/web_socket.cc
 ${ROOT}/main/harness/json.cc ${ROOT}/main/harness/transport.cc ${CJSON_ROOT}/cJSON.c)
target_include_directories(ws_transport_tests PRIVATE ${ROOT}/main ${CJSON_ROOT})
target_include_directories(ws_transport_tests PRIVATE stubs ${ROOT}/components/esp-ml307/include)
target_compile_options(ws_transport_tests PRIVATE -UNDEBUG -Wall -Wextra)
option(ENABLE_SANITIZERS "Address and undefined behavior sanitizers" ON)
if(ENABLE_SANITIZERS)
 target_compile_options(ws_transport_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(ws_transport_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME websocket_transport COMMAND ws_transport_tests)
